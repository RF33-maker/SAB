with opponent_values as (
  select
    cur.id,

    coalesce(sum(
      coalesce(opp.oreb, 0)::numeric
      * (
        greatest(
          0,
          least(cur.end_game_secs, opp.end_game_secs)
          - greatest(cur.start_game_secs, opp.start_game_secs)
        )::numeric
        / nullif(opp.seconds_played, 0)
      )
    ), 0) as opp_oreb,

    coalesce(sum(
      coalesce(opp.dreb, 0)::numeric
      * (
        greatest(
          0,
          least(cur.end_game_secs, opp.end_game_secs)
          - greatest(cur.start_game_secs, opp.start_game_secs)
        )::numeric
        / nullif(opp.seconds_played, 0)
      )
    ), 0) as opp_dreb,

    coalesce(sum(
      coalesce(opp.fg2_attempted, 0)::numeric
      * (
        greatest(
          0,
          least(cur.end_game_secs, opp.end_game_secs)
          - greatest(cur.start_game_secs, opp.start_game_secs)
        )::numeric
        / nullif(opp.seconds_played, 0)
      )
    ), 0) as opp_fg2_attempted

  from lineup_stints_context_v1 cur

  left join lineup_stints_context_v1 opp
    on opp.game_key = cur.game_key
   and opp.period = cur.period
   and opp.team_id <> cur.team_id
   and opp.start_game_secs < cur.end_game_secs
   and opp.end_game_secs > cur.start_game_secs
   and opp.seconds_played > 0

  where cur.game_key = %(game_key)s
    and cur.seconds_played > 0

  group by cur.id
)
