Important data rules:
- score_diff and team_score/opp_score in live_events are not reliable.
- Score margin must be reconstructed from lineup_stints:
  score_margin_at_stint_start =

- player_stats.player_id is uuid, lineup_stints.lineup_player_ids is text[], so use:
  s.player_id::text = any(ls.lineup_player_ids)

Garbage time rules:
Use HoopsJunkie / Cleaning The Glass style logic adapted to 40-minute games:
- Q4 starts at 1800 game seconds.
- 10:00–7:31 left Q4 = start_game_secs 1800–1949, garbage if abs margin >= 25
- 7:30–5:01 left Q4 = start_game_secs 1950–2099, garbage if abs margin >= 20
- 5:00–0:00 left Q4 = start_game_secs >= 2100, garbage if abs margin >= 10
- Period > 4 is not garbage by default.
- Exclude short-clock end-period stints:
  period 1 start_game_secs >= 598
  period 2 start_game_secs >= 1198

  abs_score_margin,
  is_garbage_time,
  is_short_clock_end_period,
  is_competitive_stint,
  context_created_at,
  context_updated_at
)
with starters as (
  select
    game_key,
    team_id,
    player_id
  from player_stats
  where game_key = %(game_key)s
    and starter = true
    and player_id is not null
),

base as (
  select
    ls.*,

    (
      select count(*)
      from starters s
      where s.game_key = ls.game_key
        and s.team_id = ls.team_id
        and s.player_id::text = any(ls.lineup_player_ids)
    ) as team_starters_on_court

  from lineup_stints ls
  where ls.game_key = %(game_key)s
),

with_margin as (
  select
    b.*,

    coalesce(
      sum(b.points_for - b.points_against) over (
        partition by b.game_key, b.team_id
        order by b.start_game_secs, b.start_action
        rows between unbounded preceding and 1 preceding
      ),
      0
    ) as score_margin_at_stint_start

  from base b
),

flagged as (
  select
    wm.*,

    abs(wm.score_margin_at_stint_start) as abs_score_margin,

    case
      when wm.period > 4 then false

      when wm.period = 4
       and wm.start_game_secs between 1800 and 1949
       and abs(wm.score_margin_at_stint_start) >= 25
      then true

      when wm.period = 4
       and wm.start_game_secs between 1950 and 2099
       and abs(wm.score_margin_at_stint_start) >= 20
      then true

      when wm.period = 4
       and wm.start_game_secs >= 2100
       and abs(wm.score_margin_at_stint_start) >= 10
      then true

      else false
    end as is_garbage_time,

    case
      when wm.period in (1, 2, 3)
       and (
         (wm.period = 1 and wm.start_game_secs >= 598)
         or (wm.period = 2 and wm.start_game_secs >= 1198)
         or (wm.period = 3 and wm.start_game_secs >= 1798)
       )
      then true
      else false
    end as is_short_clock_end_period

  from with_margin wm
)

select
  f.id,
  f.game_key,
  f.league_id,
  f.team_id,
  f.lineup_key,
  f.lineup_player_ids,
  f.lineup_names,

  f.possessions_against,
  f.is_valid_lineup,
  f.created_at,
  f.team_starters_on_court,
  f.score_margin_at_stint_start,
  f.abs_score_margin,
  f.is_garbage_time,
  f.is_short_clock_end_period,

  case
    when f.is_valid_lineup = true
     and f.is_garbage_time = false
     and f.is_short_clock_end_period = false
    then true
    else false
  end as is_competitive_stint,

  now() as context_created_at,
  now() as context_updated_at

from flagged f;
